const keyHeader = { name: 'Idempotency-Key', required: true, description: '8-128 chars' };
@ApiTags('booking') @Controller()
export class BookingController {
constructor(private readonly booking: BookingService) {}
// 1. Khách hàng gửi yêu cầu giữ chỗ có thời hạn 5 phút
@ApiBearerAuth() @Post('booking-holds') @ApiHeader(keyHeader)
hold(@CurrentUser() user: Principal, @Body() dto: SlotDto, @Headers('idempotency-key') key?: string) {
return this.booking.hold(user.userId, dto, key);
}
// 2. Chuyển đổi Hold thành Đơn đặt sân chính thức
@ApiBearerAuth() @Post('bookings') @ApiHeader(keyHeader)
create(@CurrentUser() user: Principal, @Body() dto: BookingDto,
@Req() req: AuthRequest, @Headers('idempotency-key') key?: string) {
return this.booking.create(user.userId, dto.holdId, key, req.requestId);
}
}
