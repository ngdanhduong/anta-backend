@Injectable()
export class CatalogService {
constructor(private readonly db: Database, private readonly media: MediaService) {}
// Cung cấp dữ liệu sân bãi & bảo vệ bằng 404 NOT_FOUND
async yard(tx: DbClient, id: string, lock = false) {
const yard = (await tx.query(`SELECT y.*, b.timezone, b.opening_hour, b.closing_hour
FROM catalog.yards y JOIN catalog.branches b ON b.id=y.branch_id
WHERE y.id=$1 ${lock ? 'FOR UPDATE OF y' : ''}`, [id])).rows[0];
// NẾU SÂN KHÔNG TỒN TẠI HOẶC KHÔNG ACTIVE -> 404 NOT FOUND
invariant(yard?.active, 'NOT_FOUND', 'Active yard not found', 404);
return yard;
}
}
