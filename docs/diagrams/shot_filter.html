@Catch()
export class ApiExceptionFilter implements ExceptionFilter {
catch(exception: unknown, host: ArgumentsHost) {
const req = host.switchToHttp().getRequest<AuthRequest>();
const res = host.switchToHttp().getResponse<Response>();
const pgCode = (exception as { code?: string })?.code;
let status = exception instanceof HttpException ? exception.getStatus() : 500;
let code = exception instanceof ApiError ? exception.code : status === 404 ? 'NOT_FOUND' : 'INTERNAL_ERROR';
// BẮT LỖI XUNG ĐỘT KHUNG GIỜ POSTGRES GIST CONSTRAINT (23P01) -> 409 CONFLICT
if (pgCode === '23P01') {
status = 409; code = 'SLOT_UNAVAILABLE'; message = 'The selected slot is no longer available';
}
res.status(status).json({ data: null, meta: { requestId: req.requestId }, error: { code, message } });
}
}
