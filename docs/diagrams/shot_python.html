router = APIRouter(prefix="/api/v1", tags=["catalog"])
@router.get("/public/branches")
async def branches(
request: Request,
lat: Annotated[float | None, Query(ge=-90, le=90)] = None,
lng: Annotated[float | None, Query(ge=-180, le=180)] = None,
radius: Annotated[int | None, Query(ge=100)] = None,
conn: AsyncConnection = Depends(connection)
):
require((lat is None) == (lng is None), "VALIDATION_FAILED", "lat and lng required together", 400)
# TÍNH TOÁN KHOẢNG CÁCH POSTGIS KHÔNG GIAN BẤT ĐỒNG BỘ
point = "ST_SetSRID(ST_MakePoint(%s,%s),4326)::geography"
cur = await conn.execute(f"""SELECT {CARD}, ST_Distance(b.location,{point}) AS distance_meters
FROM catalog.branches b WHERE ST_DWithin(b.location,{point},%s)""", (lng, lat, lng, lat, radius))
return envelope(request, await cur.fetchall())
