@Injectable()
export class BookingService {
constructor(
private readonly db: Database,
private readonly catalog: CatalogService, // Inject CatalogService (Provider)
private readonly idem: Idempotency,
private readonly refunds: RefundService
) {}
async quote(dto: SlotDto, tx: DbClient = this.db.pool) {
const slot = validateSlot(dto.startsAt, dto.endsAt);
// GỌI SANG CATALOG SERVICE XÁC THỰC SÂN & LẤY GIÁ THỜI GIAN THỰC
const yard = await this.catalog.yard(tx, dto.yardId);
// Kiểm tra giờ mở cửa của chi nhánh theo múi giờ
const open = (await tx.query(`SELECT ...`, [slot.start, slot.end, yard.timezone])).rows[0].valid;
invariant(open, 'OUTSIDE_OPENING_HOURS', 'Slot is outside branch opening hours', 400);
return { subtotal: yard.hourly_price * slot.hours, total: yard.hourly_price * slot.hours, currency: yard.currency };
}
}
